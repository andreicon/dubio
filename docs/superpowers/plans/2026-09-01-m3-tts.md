# Milestone 3 — TTS Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Precondition (GATE):** The M0 Fish S2 Pro Romanian evaluation (`docs/tts-engines.md`) must show acceptable results, or another TTS adapter must be selected, before running real synthesis. Logic here is validated with `FakeTTS` regardless.

**Goal:** Synthesize each utterance independently into `audio/tts/<utt>.wav` using per-character voice profiles, with deterministic caching, parallel GPU-aware scheduling, and duration matching/reporting.

**Architecture:** Voice profiles (YAML, independent from speaker IDs) map characters → voices. A cache keyed by TTS inputs prevents redundant synthesis and invalidates when translation changes. A scheduler runs independent utterances concurrently with retries and failure isolation. Duration matching compares generated vs target and records PASS/WARN/FAIL, preferring linguistic/rate correction over time-stretch.

**Tech Stack:** existing manifest/engine/timing modules, `concurrent.futures`, hashlib.

**Spec:** `PRD.md` (§17 voice profiles, §18 per-utterance synth, §19 duration matching, §32 resumability, §33 caching, §34 parallelism).

## Global Constraints

(See master plan.) M3 focus: TTS is disposable (only `engines/tts/*` know the engine); per-utterance addressing; cache invalidation on translation change; never exhaust VRAM (configurable workers); a failed utterance must not invalidate successful ones (PRD §43 reliability).

**Consumes:** M0 `TTSEngine/VoiceProfile/AudioArtifact/FakeTTS`; M1 `Manifest/Utterance/ProjectPaths`, `timing.{target_duration,duration_status}`; M2 approved `translation.text`.

from pathlib import Path
import yaml
from dubio.engines.tts.base import VoiceProfile
from dubio.errors import DubError

def load_voice_profile(path) -> VoiceProfile:
    d = yaml.safe_load(Path(path).read_text(encoding="utf-8"))
    p = d.get("parameters", {})
    return VoiceProfile(id=d["id"], engine=d["engine"], reference=d.get("reference"),
        pitch=p.get("pitch", 0), speaking_rate=p.get("speaking_rate", 1.0),
        gain_db=p.get("gain_db", 0), style=d.get("style", {}))

def resolve_voice(manifest, utterance) -> VoiceProfile:
    ch = manifest.characters.get(utterance.speaker)
    if not ch or not ch.voice:
        raise DubError("VOICE-001", f"No voice for speaker {utterance.speaker}",
                       {"utt": utterance.id}, "Assign a character voice via dubio voices")
    v = manifest.voices.get(ch.voice)
    if not v:
        raise DubError("VOICE-002", f"Voice profile missing: {ch.voice}")
    return VoiceProfile(id=ch.voice, engine=v.engine, reference=v.reference,
        pitch=v.pitch, speaking_rate=v.speaking_rate, gain_db=v.gain_db)
```

# hashing.py
import hashlib, json
def stable_hash(*parts) -> str:
    payload = json.dumps(parts, sort_keys=True, ensure_ascii=False, default=str)
    return hashlib.sha256(payload.encode("utf-8")).hexdigest()[:16]
# cache.py
from pathlib import Path
from dubio.utils.hashing import stable_hash
def tts_cache_key(engine_id, engine_version, voice_id, language, text, instructions, params) -> str:
    return stable_hash(engine_id, engine_version, voice_id, language, text, instructions, params)
class Cache:
    def __init__(self, directory): self.dir = Path(directory); self.dir.mkdir(parents=True, exist_ok=True)
    def path_for(self, key, ext="wav") -> Path: return self.dir / f"{key}.{ext}"
    def has(self, key, ext="wav") -> bool: return self.path_for(key, ext).exists()
```

```python
import shutil
from dubio.project.voices import resolve_voice
from dubio.utils.cache import tts_cache_key
from dubio.project.manifest import Manifest
from dubio.audio.measure import load_wav, duration_seconds

def synthesize_utterance(m, utt, tts, cache, paths, force: bool = False) -> None:
    voice = resolve_voice(m, utt)
    text = utt.translation.text or utt.source.text
    params = {"pitch": voice.pitch, "speaking_rate": voice.speaking_rate}
    key = tts_cache_key(tts.engine_id, tts.engine_version, voice.id, "ro", text, {}, params)
    dest = paths.tts_dir / f"{utt.id}.wav"
    dest.parent.mkdir(parents=True, exist_ok=True)
    cached = cache.path_for(key)
    if cached.exists() and not force:
        art_path, dur = cached, duration_seconds(*load_wav(cached))
    else:
        art = tts.synthesize(text, voice, "ro", {})
        shutil.copyfile(art.path, cached)
        art_path, dur = cached, art.duration
    shutil.copyfile(art_path, dest)
    utt.tts.engine = tts.engine_id
    utt.tts.engine_version = tts.engine_version
    utt.tts.voice = voice.id
    utt.tts.file = str(dest)
    utt.tts.duration = round(dur, 3)

def synthesize_project(paths, tts, config, force: bool = False) -> None:
    from dubio.utils.cache import Cache
    m = Manifest.load(paths.manifest)
    cache = Cache(paths.tts_dir / "_cache")
    for u in m.utterances:
        synthesize_utterance(m, u, tts, cache, paths, force=force)
    m.save(paths.manifest)
```

- [ ] **Step 3: Implement `utils/scheduler.py`**

```python
from dataclasses import dataclass, field
from concurrent.futures import ThreadPoolExecutor, as_completed

@dataclass
class SchedulerReport:
    succeeded: list = field(default_factory=list)
    failed: list = field(default_factory=list)

def _attempt(item, worker_fn, retries):
    last = None
    for _ in range(retries + 1):
        try:
            return True, worker_fn(item)
        except Exception as e:  # noqa: BLE001 - isolation is intentional
            last = e
    return False, last

def run_jobs(items, worker_fn, max_workers: int, retries: int = 2) -> SchedulerReport:
    report = SchedulerReport()
    with ThreadPoolExecutor(max_workers=max(1, max_workers)) as ex:
        futs = {ex.submit(_attempt, it, worker_fn, retries): it for it in items}
        for f in as_completed(futs):
            ok, result = f.result()
            if ok:
                report.succeeded.append(result)
            else:
                report.failed.append((futs[f], result))
    return report
```

```python
from dubio.pipeline.timing import duration_status, target_duration

_ORDER = ["speaking_rate", "retranslate", "time_stretch"]

def evaluate_duration(utt, cfg) -> str:
    target = target_duration(utt)
    gen = utt.tts.duration or 0.0
    status = duration_status(target, gen, cfg)
    utt.validation.duration = status
    utt.validation.measurements["duration_diff"] = round(gen - target, 3)
    utt.validation.measurements["target_duration"] = target
    utt.validation.measurements["generated_duration"] = gen
    return status

def next_correction_step(status: str, attempts_done: int) -> str | None:
    if status == "pass":
        return None
    return _ORDER[attempts_done] if attempts_done < len(_ORDER) else None
```
